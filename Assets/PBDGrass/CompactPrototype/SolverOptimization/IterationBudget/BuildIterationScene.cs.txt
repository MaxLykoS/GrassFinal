using System;
using System.IO;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using GrassFinal.PBDGrass;
using GrassFinal.PBDGrass.Editor;
using Object=UnityEngine.Object;
public static class BuildIterationScene
{
    public static void Run()
    {
        const string path="Assets/PBDGrass/Scenes/PBDGrassIterationBudget.unity";
        try
        {
            var scene=EditorSceneManager.OpenScene(GrassShowcaseSceneBuilder.ScenePath);
            var demo=Object.FindObjectOfType<GrassHandoffDemo>();
            var budget=demo.gameObject.AddComponent<GrassIterationBudgetDemo>();
            budget.Demo=demo;budget.Quality=GrassIterationBudgetDemo.Budget.Minimum2x1;
            budget.ReferenceResponse=AssetDatabase.LoadAssetAtPath<GrassResponseType>(GrassResponseBaker.TypePath(5));
            budget.MinimumResponse=AssetDatabase.LoadAssetAtPath<GrassResponseType>(GrassLowBudgetBaker.TypePath(5));
            budget.Apply();
            // Persist the unscaled value; Apply multiplies it once at runtime.
            demo.Physics.SoftBendCompliance=budget.ReferenceResponse.Physics.SoftBendCompliance;
            ConfigureSpreadLayout(Object.FindObjectOfType<GrassShowcaseController>());
            demo.gameObject.name="Grass iteration experiment - 32768 XPBD capacity";
            if(!EditorSceneManager.SaveScene(scene,path))throw new Exception("Could not save iteration scene.");
            AssetDatabase.SaveAssets();
            EditorSceneManager.OpenScene(path);
            budget=Object.FindObjectOfType<GrassIterationBudgetDemo>();demo=budget.Demo;
            budget.Apply();
            if(demo.Physics.NearSubsteps!=2 || demo.Physics.ConstraintIterations!=1 || demo.Physics.MaxActiveBlades!=32768)
                throw new Exception("Wrong serialized iteration budget.");
            demo.Initialize();
            foreach(GrassIterationBudgetDemo.Budget choice in Enum.GetValues(typeof(GrassIterationBudgetDemo.Budget)))
            {
                budget.Quality=choice;budget.Apply();
                demo.Renderer.Advance(1f/60);
                if(demo.Renderer.ActiveBladeCount>32768)throw new Exception("Pool ceiling changed.");
            }
            File.WriteAllText(Path.Combine(Environment.GetEnvironmentVariable("GRASS_ITERATION_OUTPUT"),"Results.json"),"{\"passed\":true,\"scene\":\""+path+"\"}");
            EditorApplication.Exit(0);
        }
        catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}
    }
