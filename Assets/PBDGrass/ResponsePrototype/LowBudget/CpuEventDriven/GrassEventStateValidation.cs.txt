    public static void Run()
    {
        output=Environment.GetEnvironmentVariable("GRASS_LOW_BUDGET_OUTPUT");Require(!string.IsNullOrEmpty(output),"GRASS_LOW_BUDGET_OUTPUT is required.");
        Directory.CreateDirectory(output);report=new Report();referenceRows.Clear();tolerances.Clear();
        try
        {
            report.reference=Environment.GetEnvironmentVariable("GRASS_CPU_EVENT_REFERENCE");
            report.tolerances=Environment.GetEnvironmentVariable("GRASS_CPU_EVENT_TOLERANCES")??"";
            // Exact is the default. Only individually named float channels may opt in to a tolerance.
            // Identity, packed pose/plastic, and wind remain exact; no global epsilon is accepted.
            if(!string.IsNullOrEmpty(report.tolerances))foreach(string item in report.tolerances.Split(';'))
            {
                string[] parts=item.Split('=');Require(parts.Length==2&&parts[0].StartsWith("live."),"Only named live float channels may have a tolerance: "+item);
                double value=double.Parse(parts[1],CultureInfo.InvariantCulture);Require(value>=0&&!double.IsNaN(value)&&!double.IsInfinity(value),"Invalid tolerance.");
                tolerances.Add(parts[0],value);
            }
            if(!string.IsNullOrEmpty(report.reference))
            {
                reference=JsonUtility.FromJson<Report>(File.ReadAllText(report.reference));Require(reference.passed,"Reference validation failed.");
                referenceDirectory=Path.GetDirectoryName(Path.GetFullPath(report.reference));report.compared=true;
                foreach(var row in reference.rows)referenceRows.Add(RowKey(row.scenario,row.tick,row.key),row);
            }
            foreach(string name in new[]{"fixed","events"})RunScenario(name);
            if(report.compared)Require(reference.rows.Count==report.rows.Count,"Reference/current checkpoint channel counts differ.");
            report.passed=report.failures.Count==0&&report.finite;
        }
        catch(Exception e){report.error=e.ToString();report.passed=false;Debug.LogException(e);}
        File.WriteAllText(Path.Combine(output,"Results.json"),JsonUtility.ToJson(report,true));
        Debug.Log("GRASS_EVENT_WORLD_STATE_RESULT "+JsonUtility.ToJson(report));EditorApplication.Exit(report.passed?0:1);
    }

    static void Capture(string name,int tick,PBDGrassPatchRenderer r)
    {

            Require(r.Pool.SlotForGrid(grid.GridId)*32==grid.PoolOffset&&r.Pool.IsActive(grid.GridId),"Selected identity/slot mapping is inconsistent.");

    sealed class CaptureWriter
    {
        readonly string scenario,file;readonly int tick;readonly BinaryWriter writer;readonly BinaryReader old;
        readonly int[] resident,active;bool residentComparable=true,activeComparable=true;public int count;
        public CaptureWriter(string scenario,int tick,string file,BinaryWriter writer,BinaryReader old,int[] resident,int[] active)
        {this.scenario=scenario;this.tick=tick;this.file=file;this.writer=writer;this.old=old;this.resident=resident;this.active=active;}
        public void Add(string key,Kind kind,Array values)
        {
            byte[] bytes=Bytes(values);var row=new Row{scenario=scenario,tick=tick,key=key,hash=Hash(bytes),file=file,bytes=bytes.Length,comparison=old==null?"reference":"exact"};count++;
            // All serialized float channels are finite, even in reference-only runs.
            if(kind==Kind.Float)foreach(float value in Floats(bytes))if(!Finite(value)){report.finite=false;Failure(scenario+" "+tick+" "+key+" contains a non-finite value.");break;}
            if(old!=null)
            {
                Require(old.ReadString()==key&&(Kind)old.ReadInt32()==kind,"Reference channel order/type mismatch.");int length=old.ReadInt32();
                Require(length>=0&&length<=256*1024*1024,"Corrupt reference payload length.");byte[] before=old.ReadBytes(length);Require(before.Length==length,"Truncated reference checkpoint.");
                Require(referenceRows.TryGetValue(RowKey(scenario,tick,key),out var previous),"Missing reference manifest channel.");row.referenceHash=Hash(before);
                Require(row.referenceHash==previous.hash,"Reference payload hash differs from manifest.");
                row.exact=bytes.Length==before.Length&&row.hash==row.referenceHash;
                if(key=="resident.identity")residentComparable=row.exact;
                if(key=="active.identity")activeComparable=row.exact;
                row.comparable=key.StartsWith("live.")?activeComparable:residentComparable;
                if(!row.exact)
                {
                    report.exactPassed=false;
                    if(row.comparable||kind==Kind.Identity)Compare(row,kind,before,bytes,active,resident);
                    else{row.comparison="identity mismatch; value comparison suppressed";row.withinTolerance=false;}
                    if(kind==Kind.Float&&row.comparable&&tolerances.TryGetValue(key,out double epsilon))
                    {row.tolerance=epsilon;row.withinTolerance=row.maxAbs<=epsilon;row.comparison=row.withinTolerance?"explicit channel tolerance":"outside explicit channel tolerance";}
                    else row.withinTolerance=false;
                    if(!row.withinTolerance)report.failures.Add(scenario+" tick="+tick+" "+key+" differs: "+row.firstDifference+" "+row.comparison);
                }
            }
            writer.Write(key);writer.Write((int)kind);writer.Write(bytes.Length);writer.Write(bytes);report.rows.Add(row);
        }
    }
    static void Compare(Row row,Kind kind,byte[] before,byte[] after,int[] active,int[] resident)
    {
        if(before.Length!=after.Length){row.firstDifference="channel byte length "+before.Length+" -> "+after.Length;row.comparable=false;return;}
        int first=-1;double sum=0,max=0;int differences=0;
        if(kind==Kind.Float)
        {
            var a=Floats(before);var b=Floats(after);
            for(int i=0;i<a.Length;i++)
            {
                if(BitConverter.ToInt32(before,i*4)==BitConverter.ToInt32(after,i*4))continue;
                if(first<0)first=i;differences++;double error=Finite(a[i])&&Finite(b[i])?Math.Abs((double)a[i]-b[i]):double.MaxValue;
                max=Math.Max(max,error);sum+=error*error;
            }
            row.maxAbs=max;row.rms=a.Length==0?0:Math.Sqrt(sum/a.Length);
        }
        else
        {
            var a=Words(before);var b=Words(after);for(int i=0;i<a.Length;i++)if(a[i]!=b[i]){if(first<0)first=i;differences++;}
            if(kind==Kind.Pose)PoseErrors(row,a,b);
            if(kind==Kind.Plastic)
            {
                for(int i=0;i<a.Length;i++)if(i%8<7)
                {double x=Math.Abs((double)unchecked((short)a[i])-unchecked((short)b[i]))*(2.0/32767),y=Math.Abs((double)unchecked((short)(a[i]>>16))-unchecked((short)(b[i]>>16)))*(2.0/32767);max=Math.Max(max,Math.Max(x,y));}
                row.maxAbs=max;
            }
        }
        row.firstIndex=first;row.differentComponents=differences;row.firstDifference=Describe(row.key,first,active,resident);
    }
    static void PoseErrors(Row row,uint[] before,uint[] after)
    {
        float position=0,angle=0;
        for(int blade=0;blade<before.Length/20;blade++)
        {
            for(int coordinate=0;coordinate<21;coordinate++)
            {int i=blade*20+coordinate/2,shift=(coordinate&1)*16;float a=Mathf.HalfToFloat((ushort)(before[i]>>shift)),b=Mathf.HalfToFloat((ushort)(after[i]>>shift));position=Mathf.Max(position,Mathf.Abs(a-b));}
            for(int s=0;s<7;s++){int i=blade*20+11+s;if(before[i]!=after[i])angle=Mathf.Max(angle,Vector3.Angle(UnpackDirection(before[i]),UnpackDirection(after[i])));}
        }
        row.posePositionMax=position;row.poseWidthDegreesMax=angle;
    }
    static Vector3 UnpackDirection(uint bits)
    {
        float x=(bits&65535)*(2f/65535)-1,y=(bits>>16)*(2f/65535)-1,z=1-Mathf.Abs(x)-Mathf.Abs(y);
        if(z<0){float oldX=x;x=(1-Mathf.Abs(y))*(oldX>=0?1:-1);y=(1-Mathf.Abs(oldX))*(y>=0?1:-1);}return new Vector3(x,y,z).normalized;
    }
    static string Describe(string key,int index,int[] active,int[] resident)
    {
        if(index<0)return "bit mismatch";int[] identity=key.StartsWith("live.")?active:resident;int perBlade=0,component=0,element=0,cell;
        if(key.EndsWith("identity"))cell=index/2;
        else if(key.StartsWith("history.")){cell=index/4;component=index%4;}
        else
        {
            int components=3,elements=8;
            if(key=="live.frameRotation"){components=4;elements=7;}
            else if(key=="live.frameAngular")elements=7;
            else if(key=="live.plastic"){components=4;elements=7;}
            else if(key=="live.geometry"||key=="live.normals")elements=15;
            else if(key=="compact.pose"){components=1;elements=20;}
            else if(key=="compact.plastic"){components=1;elements=8;}
            perBlade=components*elements;cell=index/(32*perBlade);component=index%components;element=(index/components)%elements;
        }
        if(cell*2+1>=identity.Length)return "index="+index+" outside matched identities";
        string where="world=("+identity[cell*2]+","+identity[cell*2+1]+")";
        if(perBlade>0)where+=" leaf="+((index/perBlade)%32)+" element="+element;return where+" component="+component+" index="+index;
    }

            var before=ModeGridState(r,result.marker);result.phaseBefore=before.Phase;Require(before.Phase==2,"Mode switch did not occur during a live exit fade.");
            result.initialSlot=r.Pool.SlotForGrid(result.marker);result.initialSeeds=r.Handoff.EntryInitializations;
            var nodes=ModeSlice<BoneInfo>(r,"bones",result.initialSlot*32*8,32*8);
            var frames=ModeSlice<GrassBladeData.Frame>(r,"frames",result.initialSlot*32*7,32*7);
            var plastic=ModeSlice<Vector4>(r,"plasticBends",result.initialSlot*32*7,32*7);
            var display=new uint[32*20];r.Compact.Pose.GetData(display,0,result.marker*32*20,display.Length);
            result.weightBefore=ModeWeight(before,r.SimulationTime,demo.Physics.ContactFadeSeconds,demo.Handoff.TransitionSeconds);
            demo.Physics.ContactDrivenSimulation=false;r.RefreshSimulationSelection();var after=ModeGridState(r,result.marker);
            result.finalSlot=r.Pool.SlotForGrid(result.marker);result.seedsAfterSwitch=r.Handoff.EntryInitializations;result.phaseAfter=after.Phase;
            result.slotPreserved=result.initialSlot==result.finalSlot&&r.Pool.IsActive(result.marker);
            Require(result.slotPreserved,"Mode switch reassigned the retained live solver slot.");
            result.nodesPreserved=Hash(Bytes(nodes))==Hash(Bytes(ModeSlice<BoneInfo>(r,"bones",result.finalSlot*32*8,32*8)));
            result.framesPreserved=Hash(Bytes(frames))==Hash(Bytes(ModeSlice<GrassBladeData.Frame>(r,"frames",result.finalSlot*32*7,32*7)));
            result.plasticPreserved=Hash(Bytes(plastic))==Hash(Bytes(ModeSlice<Vector4>(r,"plasticBends",result.finalSlot*32*7,32*7)));
            var displayAfter=new uint[display.Length];r.Compact.Pose.GetData(displayAfter,0,result.marker*32*20,displayAfter.Length);
            result.displayPreserved=Hash(Bytes(display))==Hash(Bytes(displayAfter));
            result.weightAfter=ModeWeight(after,r.SimulationTime,demo.Physics.ContactFadeSeconds,demo.Handoff.TransitionSeconds);
            result.weightDelta=Mathf.Abs(result.weightBefore-result.weightAfter);
            Require(result.nodesPreserved&&result.framesPreserved&&result.plasticPreserved&&result.displayPreserved,"Mode switch changed physical or visible state before any simulation tick.");
            Require(after.Phase<0&&result.weightDelta<2e-6f,"Mode switch did not preserve the displayed blend weight.");
            // Other near cells can legitimately seed when fixed ownership fills its
            // budget. The retained probe's state, rather than the global seed counter,
            // proves that this cell was not reset or reseeded.
            for(int i=0;i<Mathf.CeilToInt(demo.Handoff.TransitionSeconds*60)+4;i++)r.Advance(1f/60);
            result.resumeCompleted=ModeGridState(r,result.marker).Phase==0;Require(result.resumeCompleted,"A fixed-mode grid retained a permanent exit phase/history requirement.");
            demo.Physics.ContactDrivenSimulation=true;r.RefreshSimulationSelection();
            for(int i=0;i<190;i++)r.Advance(1f/60);
            result.drained=r.ActiveBladeCount==0;result.noRequiredHistory=r.Handoff.TransitionBlades==0&&r.Handoff.HistoryResidentBlades==0;
            var tiered=Field(r.Handoff,"tiered");var deadlines=Field(tiered,"eventDeadlinesHeap");
            result.noDeadlines=(int)deadlines.GetType().GetProperty("Count").GetValue(deadlines)==0;
            Require(result.drained&&result.noRequiredHistory&&result.noDeadlines,"Returning to event ownership leaked live slots, required history, or transition deadlines.");
            result.ticks=r.SimulationTicks;result.passed=true;
        }
        catch(Exception e){result.error=e.ToString();Debug.LogException(e);}
        finally{if(demo!=null)Release(demo);}
        File.WriteAllText(Path.Combine(folder,"Results.json"),JsonUtility.ToJson(result,true));EditorApplication.Exit(result.passed?0:1);
    }
